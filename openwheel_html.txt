    const posicoesPorVolta = {{ resultado.posicoes_por_volta | tojson }};
    let voltaIndex = 0;
    let playing = false;
    let playInterval = null;
    function formatarTempo(s) {
        const m = Math.floor(s / 60);
        const seg = (s % 60).toFixed(3).padStart(6, "0");
        return `${m}:${seg}`;
    }
    function formatarDiferenca(pos, dif, abandonou) {
        if (abandonou) return "ABANDONOU";
        if (pos === 1) return "-";
        if (dif === null || dif === undefined) return "-";
        if (Math.abs(dif) >= 60) {
            const m = Math.floor(Math.abs(dif) / 60);
            const s = (Math.abs(dif) % 60).toFixed(3).padStart(6, "0");
            return `+${m}:${s}`;
        }
        return `+${dif.toFixed(3)}s`;
    }
    function renderVolta(index) {
        const volta = posicoesPorVolta[index];
        document.getElementById("volta-atual").textContent = index + 1;
        const tabela = document.getElementById("tabela-replay");
        tabela.innerHTML = "<tr><th>Posição</th><th>Equipe</th><th>Tempo</th><th>DL</th><th>DF</th></tr>";
        volta.forEach(linha => {
            const tr = document.createElement("tr");
            tr.innerHTML = `<td>P${linha.posicao}</td><td>${linha.equipe}</td>` +
                            `<td>${formatarTempo(linha.tempo_acumulado)}</td>` +
                            `<td>${formatarDiferenca(linha.posicao, linha.diferenca_lider, linha.abandonou)}</td>` +
                            `<td>${formatarDiferenca(linha.posicao, linha.diferenca_frente, linha.abandonou)}</td>`;
            tabela.appendChild(tr);
        });
    }
    document.getElementById("btn-proxima").addEventListener("click", () => {
        if (voltaIndex < posicoesPorVolta.length - 1) { voltaIndex++; renderVolta(voltaIndex); }
    });
    document.getElementById("btn-anterior").addEventListener("click", () => {
        if (voltaIndex > 0) { voltaIndex--; renderVolta(voltaIndex); }
    });
    document.getElementById("btn-play").addEventListener("click", function() {
        playing = !playing;
        this.textContent = playing ? "⏸ Pausar" : "▶ Play";
        if (playing) {
            playInterval = setInterval(() => {
                if (voltaIndex < posicoesPorVolta.length - 1) {
                    voltaIndex++;
                    renderVolta(voltaIndex);
                } else {
                    clearInterval(playInterval);
                    playing = false;
                    this.textContent = "▶ Play";
                }
            }, 400);
        } else {
            clearInterval(playInterval);
        }
    });
    renderVolta(0);
</script>

<script>
    const selectOptionsHTML = `
        {% for m in modelos_disponiveis %}
        <option value="{{ m }}">{{ m }} — {% if m <= 500 %}🌤️ Seco{% elif m <= 700 %}🌧️ Molhada{% else %}⛈️ Encharcada{% endif %}</option>
        {% endfor %}

    function getPneuColorClass(modelo) {
        if (modelo <= 500) return 'pneu-seco';
        if (modelo <= 700) return 'pneu-molhado';
        return 'pneu-encharcado';
    }
    
    function getPneuColorBorder(modelo) {
        if (modelo <= 500) return '#f1c40f';
        if (modelo <= 700) return '#2ecc71';
        return '#3498db';
    }

    function addStint() {
        const container = document.getElementById('stintsContainer');
        const div = document.createElement('div');
        div.className = 'stint-card stint-row';
        div.innerHTML = `
            <div style="flex:1;">
                <span style="color:#8c9bb3; font-size:12px; display:block; margin-bottom:5px;">STINT <span class="stint-number">X</span></span>
                <select name="modelo_pneu[]" class="pneu-select" onchange="updateUI()" style="width:100%;">
                    ${selectOptionsHTML}
                </select>
            </div>
            <div>
                <span style="color:#8c9bb3; font-size:12px; display:block; margin-bottom:5px;">Voltas Alvo</span>
                <input type="number" name="voltas[]" value="15" min="1" max="100" class="input-voltas" onchange="updateUI()" style="width:80px;">
            </div>
            <div>
                <span style="color:#8c9bb3; font-size:12px; display:block; margin-bottom:5px;">Combustível (L)</span>
                <input type="number" name="combustivel_litros[]" value="40" min="1" max="150" step="0.5" class="input-combustivel" onchange="updateUI()" style="width:100px;">
            </div>
            <div style="align-self: flex-end;">
                <button type="button" class="btn-remover-stint" onclick="removeStint(this)">✕</button>
            </div>
        `;
        container.appendChild(div);
        updateUI();
    }

    function removeStint(btn) {
        const rows = document.querySelectorAll('.stint-row');
        if (rows.length <= 1) {
            alert('A estratégia deve ter pelo menos 1 stint.');
            return;
        }
        btn.closest('.stint-row').remove();
        updateUI();
    }

    function updateUI() {
        const rows = document.querySelectorAll('.stint-row');
        let totalVoltas = 0;
        let totalCombustivel = 0;
        const timeline = document.getElementById('timelineBar');
        timeline.innerHTML = '';

        rows.forEach((row, index) => {
            row.querySelector('.stint-number').innerText = index + 1;
            const voltas = parseInt(row.querySelector('.input-voltas').value) || 0;
            const combustivel = parseFloat(row.querySelector('.input-combustivel').value) || 0;
            const modeloPneu = parseInt(row.querySelector('.pneu-select').value) || 50;
            
            totalVoltas += voltas;
            totalCombustivel += combustivel;
            row.style.borderLeftColor = getPneuColorBorder(modeloPneu);
        });

        rows.forEach((row, index) => {
            const voltas = parseInt(row.querySelector('.input-voltas').value) || 0;
            const modeloPneu = parseInt(row.querySelector('.pneu-select').value) || 50;
            let perc = totalVoltas > 0 ? (voltas / totalVoltas) * 100 : 0;
            
            if (perc > 0) {
                const block = document.createElement('div');
                block.className = `timeline-stint ${getPneuColorClass(modeloPneu)}`;
                block.style.width = `${perc}%`;
                block.innerText = voltas;
                timeline.appendChild(block);
            }
        });

        document.getElementById('sumVoltasPlanejadas').innerText = totalVoltas;
        document.getElementById('sumCombustivel').innerText = totalCombustivel.toFixed(1) + ' L';
        document.getElementById('sumParadas').innerText = rows.length - 1;

        // VALIDAÇÃO DE VOLTAS E TRAVA DE BOTÃO
        const pistaVoltasText = document.getElementById('sumPistaVoltas').innerText;
        const pistaVoltas = isNaN(parseInt(pistaVoltasText)) ? 0 : parseInt(pistaVoltasText);
        const validationMsg = document.getElementById('lapValidationMsg');
        const btnAdd = document.querySelector('.btn-add-stint');

        if (pistaVoltas > 0) {
            if (totalVoltas > pistaVoltas) {
                validationMsg.innerText = `⚠️ ${totalVoltas - pistaVoltas} voltas a mais que a corrida`;
                validationMsg.style.color = '#e74c3c';
            } else if (totalVoltas < pistaVoltas) {
                validationMsg.innerText = `⚠️ Faltam ${pistaVoltas - totalVoltas} voltas`;
                validationMsg.style.color = '#f1c40f';
            } else {
                validationMsg.innerText = `✅ Voltas batem certinho`;
                validationMsg.style.color = '#2ecc71';
            }

            if (totalVoltas >= pistaVoltas) {
                btnAdd.disabled = true;
                btnAdd.innerText = "LIMITE DE VOLTAS ATINGIDO";
                btnAdd.style.opacity = "0.5";
                btnAdd.style.cursor = "not-allowed";
            } else {
                btnAdd.disabled = false;
                btnAdd.innerText = "+ ADICIONAR STINT / PIT STOP";
                btnAdd.style.opacity = "1";
                btnAdd.style.cursor = "pointer";
            }
        }
    }

    document.addEventListener('DOMContentLoaded', updateUI);
</script>
